import React, { useState, useEffect } from 'react';
import * as XLSX from 'xlsx';
import { HotTable } from '@handsontable/react-wrapper';
import 'handsontable/styles/ht-theme-classic.min.css';
import { registerAllModules } from 'handsontable/registry';
import useBaseUrl from "@docusaurus/useBaseUrl"

{registerAllModules()}

export function ExcelMdxViewer() {
  const [workbookData, setWorkbookData] = useState(null);
  const [sheetNames, setSheetNames] = useState([]);
  const [activeSheet, setActiveSheet] = useState('');
  const [gridData, setGridData] = useState([]);
  const xlsxfile = useBaseUrl("/files/WQData_sites_with_Lake_info_filtered.xlsx")
  // Load and parse local or remote .xlsx file on mount
  useEffect(() => {
      fetch(xlsxfile)
      .then((res) => {
        if (!res.ok) throw new Error(`HTTP error! status: ${res.status}`);
        return res.arrayBuffer();
      })
	  // .then((res) => res.arrayBuffer())
      .then((buffer) => {
        const wb = XLSX.read(buffer, { type: 'array' });
        setWorkbookData(wb);
        setSheetNames(wb.SheetNames);
        setActiveSheet(wb.SheetNames[0]);
        
        // Convert first sheet to an array of arrays
        const ws = wb.Sheets[wb.SheetNames[0]];
        setGridData(XLSX.utils.sheet_to_json(ws, { header: 1 }));
      })
	  .catch((err) => {
        setStatus(`Failed: ${err.message}`);
      });
  }, []);

  // Handle switching sheets via dropdown
  const handleSheetChange = (e) => {
    const newSheet = e.target.value;
    setActiveSheet(newSheet);
    if (workbookData) {
      const ws = workbookData.Sheets[newSheet];
      setGridData(XLSX.utils.sheet_to_json(ws, { header: 1 }));
    }
  };

  return (
    <div style={{ margin: '20px 0' }}>
      <div style={{ marginBottom: '10px' }}>
        <label htmlFor="sheet-select" style={{ marginRight: '8px', fontWeight: 'bold' }}>Active Sheet:</label>
        <select id="sheet-select" value={activeSheet} onChange={handleSheetChange}>
          {sheetNames.map((name) => (
            <option key={name} value={name}>{name}</option>
          ))}
        </select>
      </div>

      <HotTable
	  	theme="ht-theme-classic"
        data={gridData}
        rowHeaders={true}
        colHeaders={true}
        height="400px"
        licenseKey="non-commercial-and-evaluation"
      />
    </div>
  );
}

# Harmful Algal Bloom Predictions: An Introduction

CIROH has supported the cyanobacterial harmful algal bloom (cyanoHABs) 
forecasting research team at the University of Vermont (PI Asim Zia). Dr. Zia’s team has 
over 10 years of experience in the prediction of cyanoHABs using ground-based 
observational data products and meteorological and hydrological forecast products such 
as the National Water Model. This team has begun to integrate 
satellite data products into their cyanoHABs forecasting workflows. In addition, Dr. Zia 
and UVM’s Dr. Patrick Clemins led the Northeast Evaluation Testbed for Water 
Resource Applications (NETWA; https://uvm.edu/water/netwa), a CIROH-funded 
project. NETWA has created 

(1) data acquisition tools to help researchers access 
forecast and observational data products relevant to water resources prediction, 

(2) 
computational workflows for forecasting that can utilize high-performance computing to 
simulate large numbers of scenarios in parallel, 

(3) experimental protocol templates, 
and 

(4) a community of practice for the use of advanced research computing and data 
techniques to further water resource prediction research.

## Water Body Data Resources

The Water Body Data Filtering workflow begins by integrating two primary datasets:

(i) lake boundaries obtained from the North American Environmental Atlas: Lakes and Rivers dataset, 
Version 4.0, developed by the Commission for Environmental Cooperation (CEC, 2023),

(ii) in situ water-quality observations locations retrieved from the Water Quality Portal 
(waterqualitydata.us) for the period 2000–2023.

The water-quality dataset was first filtered to retain only monitoring stations with at least 
30 measurements during the study period, resulting in a dataset of 3,477 stations.

A spatial join was then performed to assign each water-quality monitoring station to the lake polygon 
in which it was located.

Following the spatial join, approximately 1,600 monitoring stations were found to be located 
within mapped lake polygons.

The resulting dataset was subsequently filtered to retain only major lakes with a surface area 
greater than 100 km², yielding water-quality observations for 57 major lakes, including 8 transboundary ones.

Finally, to ensure adequate temporal coverage and statistical robustness for subsequent analyses, 
only lakes with more than 100 water-quality samples over the study period were retained, 
resulting in a final dataset of 20 major lakes, including 5 transboundary ones.
